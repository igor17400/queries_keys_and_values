% Everything below is written o + dx, so the two panels are congruent.
%
%   dx  0.00 .. 2.10   the five row boxes   (centre o+1.05, width 2.10)
%   dx  2.30 .. 6.60   the middle region    (identical extent in both)
%   dx  7.45 .. 8.55   the five out boxes   (centre o+8.00, width 1.10)
%
% Rows, identical y in both panels, 1.30 apart, boxes 0.80 tall:
%   (1) 5.2   (2) 3.9   (3) 2.6   (4) 1.3   (5) 0.0
%
% (a) the middle region is ONE block of c4 with all 25 c2 arrows in it.
% (b) the middle region is FIVE bands of c4, one per row, never touching.
%
% Panel (b) lane content, absolute x, bars 0.28 tall centred on the row:
%   13.35 .. 13.75  bar of width  9   (0.40 long)
%   13.75 -> 14.25  arrow, W_1 tie at x = 14.00
%   14.25 .. 15.85  bar of width 36   (1.60 long, exactly 4x)
%   15.85 -> 16.35  arrow, W_2 tie at x = 16.10
%   16.35 .. 16.75  bar of width  9
% W_1 and W_2 sit below the stack with short dashed stubs that stop under
% the bottom lane. Nothing vertical crosses the lanes, so no mark in (b)
% can be mistaken for a connection between rows; "shared" is carried by
% drawing the two matrices once and by the note at y = -3.05.
%
%   y  6.70  panel titles          y  6.15  panel subtitles
%   y -0.85  the three widths      y -1.75  W_1, W_2 and the legend key
%   y -2.50  their shapes          y -3.05  the sharing note
% ================================================================

\def\oA{0.00}

\def\bh{0.14}          % half height of a width bar

\tikzset{
  rowbox/.style={draw=c1, line width=0.7pt, fill=c1, fill opacity=0.10,

    minimum height=0.8cm, font=\sffamily\small, text=c1, inner sep=2pt},
  outbox/.style={draw=c3, line width=0.7pt, fill=c3, fill opacity=0.26,

  learn/.style={draw=axiscolor, line width=1.5pt, fill=c4, text opacity=1,

  band/.style={fill=c4, draw=labelcolor, draw opacity=0.30, line width=0.6pt,
    rounded corners=4pt},
  mesh/.style={-{Stealth[length=3.6pt]}, line width=0.55pt, c2, opacity=0.48},
  lane/.style={-{Stealth[length=4.5pt]}, line width=0.9pt, c1, opacity=0.80},
  tie/.style={labelcolor, opacity=0.75, line width=0.7pt, dash pattern=on 2pt off 2.5pt},
  ttl/.style={font=\sffamily\large\bfseries, text=axiscolor, anchor=west},
  sub/.style={font=\sffamily\small, text=labelcolor, anchor=west},
  wlab/.style={font=\sffamily\small, text=labelcolor},
  shp/.style={font=\sffamily\small, text=axiscolor},
}
