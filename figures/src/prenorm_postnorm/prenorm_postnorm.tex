% ================================================================
% PLAN.  Two rows, each one variant, drawn by ONE skeleton macro so
% every element except the LN boxes is guaranteed identical.
%
% TYPE.  12pt class, as in residual_stream.  Everything the page has
% to read is \normalsize (12pt) or \large (14.4pt); nothing smaller.
% The page rescales the figure to 768px, so what matters is pt of
% text per pt of figure width: the frame is held near 24.3cm
% (~691pt) so \normalsize lands at ~13px and \large at ~16px.
% Holding that width with six units and 12pt labels is what fixes the
% unit geometry below: the sublayer box is 2.05cm wide (MultiHead at
% \normalsize measures 54.8pt) and it OVERHANGS both of its risers,
% which is the only way the pitch gets under 3.05cm.
%
% Local coordinates inside a row, y = 0 is the residual stream.
%
%   unit i taps the stream at x = T_i, pitch 3.00 inside a block,
%   extra 0.80 between blocks:
%     T = 0, 3.00, 6.80, 9.80, 13.60, 16.60
%     riser up at  T          to the box underside, y = 1.325
%                            (arrow drawn to 1.32 so its tip meets the border)
%     sublayer box centred (T + 0.475, 1.70), 2.05 x 0.75,
%       so it spans T-0.55 .. T+1.50 and hangs over both risers
%     riser down at T + 0.95  to the add at (T + 0.95, 0)
%   sublayers alternate MultiHead, FFN; two per block, three blocks.
%
%   THE ADD, one construction everywhere in this section:
%     circle r = 0.24, white fill, c1 stroke 1.0pt; then the stream is
%     re-stroked straight across the circle at its own 1.8pt weight so
%     it is the horizontal bar of the plus and is never cut; then the
%     vertical stroke, c1 1.8pt, half-length 0.18.  No glyph text.
%
%   THE ONE DISPLACED BOX
%     post-norm : LN on the stream at (T_i + 1.83, 0)      i = 0..5
%     pre-norm  : LN on the riser   at (T_i,       0.68)   i = 0..5
%   nothing else moves.
%     clearances in (a): add edge T+1.19 | LN T+1.43..T+2.23 |
%                        next box edge T+2.45 | next tap T+3.00
%     clearances in (b): stream 0 | LN 0.33..1.03 | box under 1.325
%
%   block panels  x: -0.85..5.55 | 5.95..12.35 | 12.75..19.15
%                 y: -1.90..2.35,  label at y = 2.45
%   stream        x: -1.25..20.90,  output head box centred 21.75
%   gradient      y = -1.05, running right to left and drawn last, so
%                 where it bumps up to y = -0.21 it is visibly threaded
%                 through the LN box rather than stopped under it:
%                 post-norm 1.83 4.83 8.63 11.63 15.43 18.43  (six)
%                 pre-norm  20.00 only, the final LN           (one)
%
%   rows: post-norm at y = 6.80, pre-norm at y = 0, titles at yb + 3.50
%   legend under row (b) at y = -2.60, centred on the frame: the add
%   key (echoing residual_stream) and what LN stands for.
%
%   ALL TEXT IS BLACK.  c2 is kept for the gradient path and for the
%   stroke and fill of the sublayer boxes only.
% ================================================================

\tikzset{
  stream/.style={c1, line width=1.8pt, line cap=round},
  branch/.style={c1, opacity=0.55, line width=1.0pt, rounded corners=4pt,
                 -{Stealth[length=5pt]}},
  sub/.style={draw=c2, line width=1.0pt, fill=c2, fill opacity=0.10,
              text opacity=1, text=black, rounded corners=3pt,
              minimum width=2.05cm, minimum height=0.75cm, inner sep=1pt,
              font=\sffamily\normalsize},
  ln/.style={draw=axiscolor, line width=0.9pt, fill=c4, text=black,
             rounded corners=2.5pt, minimum width=0.80cm,
             minimum height=0.70cm, inner sep=1pt, font=\sffamily\normalsize},
  grad/.style={c2, line width=1.4pt, dash pattern=on 3.6pt off 2.4pt,
               rounded corners=3pt, -{Stealth[length=7pt]}},
  panel/.style={fill=c4, draw=c1, draw opacity=0.30, line width=0.7pt,
                rounded corners=6pt},
  blab/.style={font=\sffamily\large, text=black, anchor=south},
  ann/.style={font=\sffamily\normalsize, text=black},
  title/.style={font=\sffamily\large, text=black, anchor=west},
  key/.style={font=\sffamily\large, text=black},
}

% the add: stream passes straight through, so the identity path is
% never broken by a white fill.  Arguments: x and y of the centre.
\newcommand{\addcirc}[2]{%
  \draw[draw=c1, line width=1.0pt, fill=white] (#1, #2) circle (0.24);
  \draw[stream] ({#1 - 0.26}, #2) -- ({#1 + 0.26}, #2);
  \draw[stream] (#1, {#2 - 0.18}) -- (#1, {#2 + 0.18});
}

% ---------------------------------------------------------------- skeleton
% everything both variants share, drawn identically in both rows
\newcommand{\rowpanels}{%
  \foreach \a/\b/\n in {-0.85/5.55/1, 5.95/12.35/2, 12.75/19.15/3} {
      \path[panel] (\a, -1.90) rectangle (\b, 2.35);
      \node[blab] at ({0.5*(\a+\b)}, 2.45) {block \n};
  }
}
\newcommand{\rowbody}{%
  \draw[stream, -{Stealth[length=8pt]}] (-1.25, 0) -- (20.90, 0);
  \node[font=\sffamily\LARGE, text=c1, anchor=east] at (-1.40, 0) {$\mathbf{x}$};
  \node[ann, anchor=north] at (0.10, -0.42) {$d_{\mathrm{model}} = 9$};
  \foreach \i/\T/\nm in {0/0/MultiHead, 1/3.00/FFN, 2/6.80/MultiHead,
                         3/9.80/FFN, 4/13.60/MultiHead, 5/16.60/FFN} {
      \pgfmathsetmacro{\Tb}{\T + 0.95}
      \fill[c1] (\T, 0) circle (0.09);
      \node[sub] (s\i) at ({\T + 0.475}, 1.70) {$\mathrm{\nm}$};
      \draw[branch] (\T, 0) -- (\T, 1.32);
      \draw[branch] (\Tb, 1.325) -- (\Tb, 0.34);
      \addcirc{\Tb}{0}
  }
  \node[draw=c1, line width=0.9pt, fill=white, rounded corners=3pt,
        minimum width=1.70cm, minimum height=1.05cm, align=center,
        text=black, font=\sffamily\normalsize] at (21.75, 0) {output\\head};
  \node[ann, anchor=north west] at (-0.75, -1.20) {gradient};
}
